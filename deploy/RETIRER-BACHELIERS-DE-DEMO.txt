SELECT b.id AS fiche, b.numero_releve, b.nom, b.prenom, b.is_registered,
       u.id AS compte, u.numero_identifiant, u.role, u.created_at
FROM bacheliers b
LEFT JOIN students s ON s.bachelier_id = b.id
LEFT JOIN users u    ON u.id = s.user_id
WHERE b.numero_releve IN ('BAC2023001','BAC2023002','BAC2023003');

SELECT
  (SELECT COUNT(*) FROM student_requests sr
     JOIN students s   ON s.id = sr.student_id

  4. SUPPRESSION — transaction, rien n'est definitif avant COMMIT
-----------------------------------------------------------------------

START TRANSACTION;

DROP TEMPORARY TABLE IF EXISTS tmp_comptes;
CREATE TEMPORARY TABLE tmp_comptes (id BIGINT UNSIGNED PRIMARY KEY);
INSERT IGNORE INTO tmp_comptes (id)
  SELECT u.id FROM users u
  JOIN students s   ON s.user_id = u.id
  JOIN bacheliers b ON b.id = s.bachelier_id
  WHERE b.numero_releve IN ('BAC2023001','BAC2023002','BAC2023003')
    AND u.role IN ('etudiant','candidat');

SELECT COUNT(*) AS comptes_vises FROM tmp_comptes;

UPDATE bacheliers SET is_registered = 0
WHERE numero_releve IN ('BAC2023001','BAC2023002','BAC2023003');

DELETE p FROM paiements p
JOIN student_requests sr ON sr.id = p.payable_id
JOIN students s          ON s.id = sr.student_id
JOIN tmp_comptes t       ON t.id = s.user_id
WHERE p.payable_type = 'App\\Models\\StudentRequest';

DELETE mr FROM model_has_roles mr
JOIN tmp_comptes t ON t.id = mr.model_id
WHERE mr.model_type = 'App\\Models\\User';

DELETE mp FROM model_has_permissions mp
JOIN tmp_comptes t ON t.id = mp.model_id
WHERE mp.model_type = 'App\\Models\\User';

DELETE se FROM sessions se JOIN tmp_comptes t ON t.id = se.user_id;

DELETE u FROM users u JOIN tmp_comptes t ON t.id = u.id;

SELECT
  (SELECT COUNT(*) FROM students s JOIN bacheliers b ON b.id = s.bachelier_id
     WHERE b.numero_releve IN ('BAC2023001','BAC2023002','BAC2023003'))  AS comptes_demo_restants,
  (SELECT COUNT(*) FROM bacheliers WHERE is_registered = 1
     AND numero_releve IN ('BAC2023001','BAC2023002','BAC2023003'))      AS fiches_demo_inscrites,
  (SELECT COUNT(*) FROM model_has_roles mr WHERE mr.model_type = 'App\\Models\\User'
     AND NOT EXISTS (SELECT 1 FROM users u WHERE u.id = mr.model_id))    AS roles_orphelins,
  (SELECT COUNT(*) FROM users WHERE role NOT IN ('etudiant','candidat')) AS personnel,
  (SELECT COUNT(*) FROM anciens_etudiants)                               AS referentiel_anciens,
  (SELECT COUNT(*) FROM request_types)                                   AS pieces_configurees;
